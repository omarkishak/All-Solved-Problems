/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* insertionSortList(ListNode* head) {
        ListNode* dummy_ptr = new ListNode();
        ListNode* current_ptr = head;

        while (current_ptr != nullptr) {
            ListNode* prev = dummy_ptr;

            //find correct position
            while (prev->next != nullptr && prev->next->val < current_ptr->val) {
                prev = prev->next;
            }

            ListNode* nextNode = current_ptr->next;

            //insert current node
            current_ptr->next = prev->next;
            prev->next = current_ptr;

            current_ptr = nextNode;
        }

        return dummy_ptr->next;
    }
};
